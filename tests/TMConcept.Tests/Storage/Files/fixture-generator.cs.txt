// General step defaults (step_models): two are removed by the 0.1.5 upgrade, two become each profile's choice in 0.1.6.

    new() { Step = step, Provider = provider, Model = model, KeyId = keyId, OptionsJson = options };
var generals = new[]
{

#if V1

{
    await db.Database.EnsureCreatedAsync();
    db.ApiKeys.Add(apiKey);
    db.Profiles.AddRange(profile1, profile2);
    db.ProfileImages.Add(image);
    db.Concepts.AddRange(concept1, concept2);

{

    var none = CancellationToken.None;
    await ApiKeyTable.Instance.InsertAsync(db, apiKey, none);
    await ProfileTable.Instance.InsertAsync(db, profile1, none);
    await ProfileTable.Instance.InsertAsync(db, profile2, none);
    await ProfileImageTable.Instance.InsertAsync(db, image, none);

    await VideoCostTable.Instance.InsertAsync(db, cost2, none);

#if V2
    // 0.1.5 had removed the image and voice defaults already (its upgrade deleted them); the two others stay until 0.1.6.
    foreach (var general in generals.Where(g => g.Step is "concepts" or "code"))
    {
        await StepModelTable.Instance.InsertAsync(db, general, none);

    // An explicit choice of profile 1 (kept by the 0.1.6 upgrade) and an explicit "off" (kept too).

#else
    await ProfileStepModelTable.Instance.InsertAsync(db, new StepModelChoice { ProfileId = p1, Step = "concepts", Provider = "openrouter", Model = "x-model", KeyId = key, OptionsJson = "{}" }, none);
    await ProfileStepModelTable.Instance.InsertAsync(db, new StepModelChoice { ProfileId = p2, Step = "code", Provider = "openai", Model = "gpt-5", OptionsJson = "{\"effort\":\"high\"}" }, none);

}
#endif
